   112→function formatDate(dateStr: string): string {
   113→  const d = new Date(dateStr)
   114→  return d.toLocaleDateString('sl-SI', {
   115→    day: 'numeric',
   116→    month: 'short',
   117→  })
   118→}
   119→
   120→function formatDateNice(dateStr: string): string {
   121→  const d = new Date(dateStr)
   122→  const now = new Date()
   123→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   124→  const date = new Date(d.getFullYear(), d.getMonth(), d.getDate())
   125→  const diffDays = Math.round((today.getTime() - date.getTime()) / 86400000)
   126→
   127→  if (diffDays === 0) return 'danes'
   128→  if (diffDays === 1) return 'včeraj'
   129→  if (diffDays === -1) return 'jutri'
   130→  if (diffDays === -2) return 'pjutri'
   131→  if (diffDays > 1 && diffDays <= 7) return `pred ${diffDays} dnevi`
   132→  if (diffDays > 7 && diffDays <= 30) return `pred ${Math.floor(diffDays / 7)} tedni`
   133→  if (diffDays < -2 && diffDays >= -7) return `čez ${Math.abs(diffDays)} dni`
   134→  return formatDate(dateStr)
   135→}
   136→
   137→function formatRelativeTime(dateStr: string): string {
   138→  const now = new Date()
   139→  const date = new Date(dateStr)
   140→  const diffMs = now.getTime() - date.getTime()
   141→  const diffMin = Math.floor(diffMs / 60000)
   142→  const diffH = Math.floor(diffMs / 3600000)
   143→  const diffD = Math.floor(diffMs / 86400000)
   144→
   145→  if (diffMin < 1) return 'Pravkar'
   146→  if (diffMin < 60) return `pred ${diffMin} min`
   147→  if (diffH < 24) return `pred ${diffH} ur`
   148→  if (diffD < 7) return `pred ${diffD} dni`
   149→  return formatDate(dateStr)
   150→}

   179→function estimateSlats(measurements: Array<{ dolzinaMm: number }>, slatWidth: number = 80, maxGap: number = 100): number {
   180→  return measurements.reduce((total, m) => {
   181→    const n = Math.ceil((m.dolzinaMm - maxGap) / (maxGap + slatWidth))
   182→    return total + n
   183→  }, 0)
   184→}

   186→function MiniRailingDiagram({ dolzina, visina }: { dolzina: number; visina: number }) {
   187→  const maxDim = Math.max(dolzina, visina)
   188→  const heightPct = Math.min((visina / maxDim) * 30, 30)
   189→  const n = Math.ceil((dolzina - 100) / (100 + 80))
   190→  const actualGap = (dolzina - n * 80) / (n + 1)
   191→  const numSlats = Math.min(n, 12)
   192→  const gapPct = actualGap / dolzina * 100
   193→
   194→  return (
   195→    <div className="relative rounded-md border border-roksal-navy/15 bg-gradient-to-b from-roksal-navy/3 to-roksal-navy/6 p-2" style={{ minHeight: `${Math.max(heightPct, 14)}px` }}>
   196→      <div className="flex items-end h-full gap-0" style={{ height: `${Math.max(heightPct, 14)}px` }}>
   197→        <div className="w-[3px] h-full bg-roksal-navy rounded-full" />
   198→        <div className="flex-1 flex items-end h-full gap-0">
   199→          <div className="h-full bg-transparent" style={{ width: `${gapPct}%` }} />
   200→          {Array.from({ length: numSlats }).map((_, i) => (
   201→            <div key={i} className="flex h-full">
   202→              <div className="h-[85%] bg-roksal-navy/70 rounded-[1px]" style={{ width: `${(80 / dolzina) * 100}%`, minWidth: '1px' }} />
   203→              {i < numSlats - 1 && (
   204→                <div className="h-full bg-roksal-amber/25" style={{ width: `${gapPct}%`, minWidth: '1px' }} />
   205→              )}
   206→            </div>
   207→          ))}
   208→          <div className="h-full bg-transparent" style={{ width: `${gapPct}%` }} />
   209→        </div>
   210→        <div className="w-[3px] h-full bg-roksal-navy rounded-full" />
   211→      </div>
   212→      <div className="mt-0.5 flex">
   213→        <div className="w-[3px] bg-roksal-navy rounded-full" />
   214→        <div className="flex-1 h-[2px] bg-roksal-navy/30 rounded" />
   215→        <div className="w-[3px] bg-roksal-navy rounded-full" />
   216→      </div>
   217→    </div>
   218→  )
   219→}
