   267→   267→  // Quick spacing calc for a measurement
   268→   268→  function getQuickSpacing(dolzinaMm: number, visinaMm: number, slatWidth: number = 80, maxGap: number = 100) {
   269→   269→    const n = Math.ceil((dolzinaMm - maxGap) / (maxGap + slatWidth))
   270→   270→    const actualGap = (dolzinaMm - n * slatWidth) / (n + 1)
   271→   271→    return {
   272→   272→      slatCount: n,
   273→   273→      gap: Math.round(actualGap * 10) / 10,
   274→   274→      compliant: actualGap <= 100,
   275→   275→      postSpacing: dolzinaMm / Math.max(1, n > 5 ? Math.ceil(n / 5) : 2),
   276→   276→    }
   277→   277→  }
   278→   278→
   279→   279→  const lidarScans = measurements.filter((m) => m.lidarScanUrl).length
   280→   280→  const totalLength = measurements.reduce((sum, m) => sum + m.dolzinaMm, 0)
   281→   281→  const avgLength = measurements.length > 0
   282→   282→    ? measurements.reduce((s, m) => s + m.dolzinaMm, 0) / measurements.length
   283→   283→    : 0
   284→   284→  const avgHeight = measurements.length > 0
   285→   285→    ? measurements.reduce((s, m) => s + m.visinaMm, 0) / measurements.length
   286→   286→    : 0
   287→   287→  const totalPosts = measurements.reduce((sum, m) => sum + (m.steviloStebrov || 0), 0)
   288→   288→
   289→   289→  // Date grouping
   290→   290→  const groupedMeasurements = useMemo((): MeasurementGroup[] => {
   291→   291→    const now = new Date()
   292→   292→    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   293→   293→    const yesterday = new Date(today.getTime() - 86400000)
   294→   294→
   295→   295→    const groups: MeasurementGroup[] = []
   296→   296→    const grouped = new Map<string, Measurement[]>()
   297→   297→
   298→   298→    const sorted = [...measurements].sort(
   299→   299→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   300→   300→    )
   301→   301→
   302→   302→    for (const m of sorted) {
   303→   303→      const mDate = new Date(m.createdAt)
   304→   304→      const mDay = new Date(mDate.getFullYear(), mDate.getMonth(), mDate.getDate())
   305→   305→
   306→   306→      let label: string
   307→   307→      if (mDay.getTime() === today.getTime()) {
   308→   308→        label = 'Danes'
   309→   309→      } else if (mDay.getTime() === yesterday.getTime()) {
   310→   310→        label = 'Včeraj'
   311→   311→      } else {
   312→   312→        label = mDate.toLocaleDateString('sl-SI', { day: 'numeric', month: 'long' })
   313→   313→      }
   314→   314→
   315→   315→      if (!grouped.has(label)) {
   316→   316→        grouped.set(label, [])
   317→   317→      }
   318→   318→      grouped.get(label)!.push(m)
   319→   319→    }
   320→   320→
   321→   321→    for (const [label, meas] of grouped) {
   322→   322→      groups.push({ label, date: new Date(meas[0].createdAt), measurements: meas })
   323→   323→    }
   324→   324→
   325→   325→    return groups
   326→   326→  }, [measurements])

   420→   420→  function handleExportCSV() {
   421→   421→    if (measurements.length === 0) {
   422→   422→      toast.error('Ni meritev za izvoz')
   423→   423→      return
   424→   424→    }
   425→   425→    const header = 'Lokacija,Dolzina(mm),Visina(mm),Stevilo stebrov,Tip podlage,Kot,Opombe,Datum'
   426→   426→    const rows = measurements.map((m) => {
   427→   427→      const lokacija = m.lokacija || ''
   428→   428→      const dolzina = String(m.dolzinaMm)
   429→   429→      const visina = String(m.visinaMm)
   430→   430→      const stebri = m.steviloStebrov ? String(m.steviloStebrov) : ''
   431→   431→      const podlaga = m.tipPodlage ? (groundTypeLabels[m.tipPodlage as GroundType] || m.tipPodlage) : ''
   432→   432→      const kot = m.kot ? String(m.kot) : ''
   433→   433→      const opombe = (m.opombe || '').replace(/"/g, '""')
   434→   434→      const datum = new Date(m.createdAt).toLocaleDateString('sl-SI')
   435→   435→      return `${lokacija},${dolzina},${visina},${stebri},${podlaga},${kot},"${opombe}",${datum}`
   436→   436→    })
   437→   437→    const csvContent = '\uFEFF' + header + '\n' + rows.join('\n')
   438→   438→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   439→   439→    const url = URL.createObjectURL(blob)
   440→   440→    const link = document.createElement('a')
   441→   441→    link.href = url
   442→   442→    link.download = `meritve_${new Date().toISOString().slice(0, 10)}.csv`
