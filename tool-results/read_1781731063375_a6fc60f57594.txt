   267→  // Quick spacing calc for a measurement
   268→  function getQuickSpacing(dolzinaMm: number, visinaMm: number, slatWidth: number = 80, maxGap: number = 100) {
   269→    const n = Math.ceil((dolzinaMm - maxGap) / (maxGap + slatWidth))
   270→    const actualGap = (dolzinaMm - n * slatWidth) / (n + 1)
   271→    return {
   272→      slatCount: n,
   273→      gap: Math.round(actualGap * 10) / 10,
   274→      compliant: actualGap <= 100,
   275→      postSpacing: dolzinaMm / Math.max(1, n > 5 ? Math.ceil(n / 5) : 2),
   276→    }
   277→  }

   289→  // Date grouping
   290→  const groupedMeasurements = useMemo((): MeasurementGroup[] => {
   291→    const now = new Date()
   292→    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   293→    const yesterday = new Date(today.getTime() - 86400000)
   294→
   295→    const groups: MeasurementGroup[] = []
   296→    const grouped = new Map<string, Measurement[]>()
   297→
   298→    const sorted = [...measurements].sort(
   299→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   300→    )
   301→
   302→    for (const m of sorted) {
   303→      const mDate = new Date(m.createdAt)
   304→      const mDay = new Date(mDate.getFullYear(), mDate.getMonth(), mDate.getDate())
   305→
   306→      let label: string
   307→      if (mDay.getTime() === today.getTime()) {
   308→        label = 'Danes'
   309→      } else if (mDay.getTime() === yesterday.getTime()) {
   310→        label = 'Včeraj'
   311→      } else {
   312→        label = mDate.toLocaleDateString('sl-SI', { day: 'numeric', month: 'long' })
   313→      }
   314→
   315→      if (!grouped.has(label)) {
   316→        grouped.set(label, [])
   317→      }
   318→      grouped.get(label)!.push(m)
   319→    }
   320→
   321→    for (const [label, meas] of grouped) {
   322→      groups.push({ label, date: new Date(meas[0].createdAt), measurements: meas })
   323→    }
   324→
   325→    return groups
   326→  }, [measurements])

   333→  function renderRailingDiagram(dolzina: number, visina: number, _slatCount?: number) {

   420→  function handleExportCSV() {
   421→    if (measurements.length === 0) {
   422→      toast.error('Ni meritev za izvoz')
   423→      return
   424→    }
   425→    const header = 'Lokacija,Dolzina(mm),Visina(mm),Stevilo stebrov,Tip podlage,Kot,Opombe,Datum'
   426→    const rows = measurements.map((m) => {
   427→      const lokacija = m.lokacija || ''
   428→      const dolzina = String(m.dolzinaMm)
   429→      const visina = String(m.visinaMm)
   430→      const stebri = m.steviloStebrov ? String(m.steviloStebrov) : ''
   431→      const podlaga = m.tipPodlage ? (groundTypeLabels[m.tipPodlage as GroundType] || m.tipPodlage) : ''
   432→      const kot = m.kot ? String(m.kot) : ''
   433→      const opombe = (m.opombe || '').replace(/"/g, '""')
   434→      const datum = new Date(m.createdAt).toLocaleDateString('sl-SI')
   435→      return `${lokacija},${dolzina},${visina},${stebri},${podlaga},${kot},"${opombe}",${datum}`
   436→    })
   437→    const csvContent = '\uFEFF' + header + '\n' + rows.join('\n')
   438→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   439→    const url = URL.createObjectURL(blob)
   440→    const link = document.createElement('a')
   441→    link.href = url
   442→    link.download = `meritve_${new Date().toISOString().slice(0, 10)}.csv`
   443→    document.body.appendChild(link)
   444→    link.click()
   445→    document.body.removeChild(link)
   446→    URL.revokeObjectURL(url)
   447→    toast.success('CSV izvožen!')
   448→  }
