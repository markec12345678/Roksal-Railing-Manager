   540→// P3 — pretvorba enot (mm/cm/m) v mm
   541→function convertToMm(value: number, unit: EnotaTip): number {
   542→  if (!Number.isFinite(value)) return 0
   543→  switch (unit) {
   544→    case 'mm':
   545→      return value
   546→    case 'cm':
   547→      return value * 10
   548→    case 'm':
   549→      return value * 1000
   550→  }
   551→}
   552→
   553→// P3 — prikaz v primarni enoti (default mm)
   554→function formatInPrimaryUnit(mm: number, primary: EnotaTip): string {
   555→  if (!Number.isFinite(mm)) return '—'
   556→  switch (primary) {
   557→    case 'mm':
   558→      return `${Math.round(mm)}mm`
   559→    case 'cm':
   560→      return `${(mm / 10).toFixed(1)}cm`
   561→    case 'm':
   562→      return `${(mm / 1000).toFixed(2)}m`
   563→  }
   564→}

   566→// P3 — izračun stopniščnih dimenzij
   567→function calculateStairDimensions(
   568→  skupnaVisinaMm: number,
   569→  stStopnic: number,
   570→  globinaStopniceMm: number,
   571→  sirinaStopniceMm?: number
   572→): StairCalc {
   573→  if (
   574→    skupnaVisinaMm <= 0 ||
   575→    stStopnic <= 0 ||
   576→    globinaStopniceMm <= 0 ||
   577→    !Number.isFinite(skupnaVisinaMm) ||
   578→    !Number.isFinite(stStopnic) ||
   579→    !Number.isFinite(globinaStopniceMm)
   580→  ) {
   581→    return {
   582→      visinaPosamezne: 0,
   583→      kotStopinje: 0,
   584→      dolzinaKosa: 0,
   585→      skupnaDolzina: 0,
   586→      priporocilo: 'Vnesite veljavne vhodne podatke',
   587→      priporociloColor: 'text-muted-foreground',
   588→      valid: false,
   589→    }
   590→  }
   591→  const visinaPosamezne = skupnaVisinaMm / stStopnic
   592→  const kotRad = Math.atan(visinaPosamezne / globinaStopniceMm)
   593→  const kotStopinje = (kotRad * 180) / Math.PI
   594→  const dolzinaKosa = Math.sqrt(visinaPosamezne ** 2 + globinaStopniceMm ** 2) * stStopnic
   595→  const rezerva = sirinaStopniceMm ? sirinaStopniceMm * 0.5 : 200 // dodaten rob
   596→  const skupnaDolzina = dolzinaKosa + rezerva
   597→  let priporocilo = 'Standardni kot 30–35°'
   598→  let priporociloColor = 'text-green-700'
   599→  if (kotStopinje > 40) {
   600→    priporocilo = 'Nevarno: >40° (prestrmo!)'
   601→    priporociloColor = 'text-red-600'
   602→  } else if (kotStopinje > 37) {
   603→    priporocilo = 'Prestrmo: >37°'
   604→    priporociloColor = 'text-orange-600'
   605→  } else if (kotStopinje < 25) {
   606→    priporocilo = 'Ploščato: <25°'
   607→    priporociloColor = 'text-amber-600'
   608→  }
   609→  return {
   610→    visinaPosamezne,
   611→    kotStopinje,
   612→    dolzinaKosa,
   613→    skupnaDolzina,
   614→    priporocilo,
   615→    priporociloColor,
   616→    valid: true,
   617→  }
   618→}

   628→// P3 — izračun števila WPC palic za dano orientacijo
   629→function calcWpcPalice(
   630→  orientacija: Segment['type'],
   631→  dolzinaMm: number,
   632→  visinaMm: number,
   633→  sirinaPalice: number,
   634→  razmikPalic: number
   635→): number {
   636→  if (sirinaPalice <= 0 || razmikPalic <= 0) return 0
   637→  const korak = sirinaPalice + razmikPalic
   638→  if (
   639→    orientacija === 'WPC_POKOCNE' ||
   640→    orientacija === 'WPC_VODORAVNE'
   641→  ) {
   642→    const relevantnaDim = orientacija === 'WPC_POKOCNE' ? dolzinaMm : visinaMm
   643→    if (relevantnaDim <= 0) return 0
   644→    return Math.max(0, Math.floor((relevantnaDim - sirinaPalice) / korak) + 1)
   645→  }
   646→  if (orientacija === 'WPC_POSEVNE') {
   647→    // mreža: (dolžina / korak) × (višina / korak) — približno
   648→    if (dolzinaMm <= 0 || visinaMm <= 0) return 0
   649→    const nDolzina = Math.max(0, Math.floor((dolzinaMm - sirinaPalice) / korak) + 1)
   650→    const nVisina = Math.max(0, Math.floor((visinaMm - sirinaPalice) / korak) + 1)
   651→    return nDolzina * nVisina
   652→  }
   653→  return 0
   654→}
