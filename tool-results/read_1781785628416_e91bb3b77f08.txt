    86→// Komprimira datoteko v base64 JPEG (max 1280px širina, kvaliteta 0.75)
    87→function compressImageFile(file: File, maxSize = 1280, quality = 0.75): Promise<string> {
    88→  return new Promise((resolve, reject) => {
    89→    const reader = new FileReader()
    90→    reader.onload = () => {
    91→      const img = new Image()
    92→      img.onload = () => {
    93→        const scale = Math.min(1, maxSize / Math.max(1, img.naturalWidth))
    94→        const canvas = document.createElement('canvas')
    95→        canvas.width = Math.max(1, Math.round(img.naturalWidth * scale))
    96→        canvas.height = Math.max(1, Math.round(img.naturalHeight * scale))
    97→        const ctx = canvas.getContext('2d')
    98→        if (!ctx) return reject(new Error('Canvas ni podprt'))
    99→        ctx.drawImage(img, 0, 0, canvas.width, canvas.height)
   100→        resolve(canvas.toDataURL('image/jpeg', quality))
   101→      }
   102→      img.onerror = () => reject(new Error('Napaka pri nalaganju slike'))
   103→      img.src = reader.result as string
   104→    }
   105→    reader.onerror = () => reject(new Error('Napaka pri branju datoteke'))
   106→    reader.readAsDataURL(file)
   107→  })
   108→}

   110→// Ocena velikosti base64 slike v bajtih
   111→function estimateBytes(base64: string): number {
   112→  const idx = base64.indexOf(',')
   113→  const b64 = idx >= 0 ? base64.slice(idx + 1) : base64
   114→  return Math.floor(b64.length * 0.75)
   115→}

   117→function formatBytes(bytes: number): string {
   118→  if (bytes < 1024) return `${bytes} B`
   119→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} KB`
   120→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   121→}
   122→
   123→function formatRelativeTime(iso: string): string {
   124→  const date = new Date(iso)
   125→  const diff = Date.now() - date.getTime()
   126→  const sec = Math.floor(diff / 1000)
   127→  if (sec < 60) return 'pravkar'
   128→  const min = Math.floor(sec / 60)
   129→  if (min < 60) return `pred ${min} min`
   130→  const h = Math.floor(min / 60)
   131→  if (h < 24) return `pred ${h} h`
   132→  const d = Math.floor(h / 24)
   133→  if (d < 7) return `pred ${d} d`
   134→  return date.toLocaleDateString('sl-SI')
   135→}
   136→
   137→function formatLength(mm: number): string {
   138→  if (mm < 1000) return `${Math.round(mm)} mm`
   139→  return `${(mm / 1000).toFixed(2)} m`
   140→}

   142→// Nariše puščično glavo
   143→function drawArrowHead(
   144→  ctx: CanvasRenderingContext2D,
   145→  x1: number, y1: number,
   146→  x2: number, y2: number,
   147→  color: string, width: number
   148→) {
   149→  const angle = Math.atan2(y2 - y1, x2 - x1)
   150→  const headLen = Math.max(10, width * 3)
   151→  ctx.strokeStyle = color
   152→  ctx.fillStyle = color
   153→  ctx.lineWidth = width
   154→  ctx.beginPath()
   155→  ctx.moveTo(x2, y2)
   156→  ctx.lineTo(x2 - headLen * Math.cos(angle - Math.PI / 6), y2 - headLen * Math.sin(angle - Math.PI / 6))
   157→  ctx.moveTo(x2, y2)
   158→  ctx.lineTo(x2 - headLen * Math.cos(angle + Math.PI / 6), y2 - headLen * Math.sin(angle + Math.PI / 6))
   159→  ctx.stroke()
   160→}

   162→// Nariše posamezno anotacijo na kontekst
   163→function drawAnnotation(ctx: CanvasRenderingContext2D, ann: Annotation) {
   164→  ctx.strokeStyle = ann.color
   165→  ctx.fillStyle = ann.color
   166→  ctx.lineWidth = Math.max(0.5, ann.width)
   167→  ctx.lineCap = 'round'
   168→  ctx.lineJoin = 'round'
   169→
   170→  if (ann.type === 'pen' && ann.points.length > 0) {
   171→    ctx.beginPath()
   172→    ctx.moveTo(ann.points[0].x, ann.points[0].y)
   173→    for (let i = 1; i < ann.points.length; i++) {
   174→      ctx.lineTo(ann.points[i].x, ann.points[i].y)
   175→    }
   176→    ctx.stroke()
   177→    return
   178→  }
   179→
   180→  if (ann.type === 'text' && ann.text && ann.points.length >= 1) {
   181→    const p = ann.points[0]
   182→    const fs = ann.fontSize ?? 18
   183→    ctx.font = `bold ${fs}px sans-serif`
   184→    ctx.textAlign = 'left'
   185→    ctx.textBaseline = 'top'
   186→    // Beli rob za vidnost
   187→    ctx.lineWidth = Math.max(2, fs * 0.18)
   188→    ctx.strokeStyle = ann.color === '#ffffff' ? '#1d2b3e' : '#ffffff'
   189→    ctx.strokeText(ann.text, p.x, p.y)
   190→    ctx.fillStyle = ann.color
   191→    ctx.fillText(ann.text, p.x, p.y)
   192→    return
   193→  }
   194→
   195→  if (ann.points.length < 2) return
   196→
   197→  const [a, b] = ann.points
   198→  if (ann.type === 'arrow') {
   199→    ctx.beginPath()
   200→    ctx.moveTo(a.x, a.y)
   201→    ctx.lineTo(b.x, b.y)
   202→    ctx.stroke()
   203→    drawArrowHead(ctx, a.x, a.y, b.x, b.y, ann.color, ann.width)
   204→  } else if (ann.type === 'line') {
   205→    ctx.beginPath()
   206→    ctx.moveTo(a.x, a.y)
   207→    ctx.lineTo(b.x, b.y)
   208→    ctx.stroke()
   209→  } else if (ann.type === 'measure') {
   210→    ctx.beginPath()
   211→    ctx.moveTo(a.x, a.y)
   212→    ctx.lineTo(b.x, b.y)
   213→    ctx.stroke()
   214→    drawArrowHead(ctx, a.x, a.y, b.x, b.y, ann.color, ann.width)
   215→    drawArrowHead(ctx, b.x, b.y, a.x, a.y, ann.color, ann.width)
   216→    if (ann.label) {
   217→      const mx = (a.x + b.x) / 2
   218→      const my = (a.y + b.y) / 2
   219→      const fs = ann.fontSize ?? 14
   220→      ctx.font = `bold ${fs}px sans-serif`
   221→      const w = ctx.measureText(ann.label).width + 12
   222→      ctx.fillStyle = ann.color
   223→      ctx.fillRect(mx - w / 2, my - fs / 2 - 4, w, fs + 8)
   224→      ctx.fillStyle = ann.color === '#ffffff' ? '#1d2b3e' : '#ffffff'
   225→      ctx.textAlign = 'center'
   226→      ctx.textBaseline = 'middle'
   227→      ctx.fillText(ann.label, mx, my)
   228→    }
   229→  } else if (ann.type === 'rect') {
   230→    const x = Math.min(a.x, b.x)
   231→    const y = Math.min(a.y, b.y)
   232→    const w = Math.abs(b.x - a.x)
   233→    const h = Math.abs(b.y - a.y)
   234→    ctx.strokeRect(x, y, w, h)
   235→  } else if (ann.type === 'circle') {
   236→    const cx = a.x
   237→    const cy = a.y
   238→    const r = Math.max(1, Math.hypot(b.x - a.x, b.y - a.y))
   239→    ctx.beginPath()
   240→    ctx.arc(cx, cy, r, 0, Math.PI * 2)
   241→    ctx.stroke()
   242→  }
   243→}

  1025→// CAMERA CAPTURE — polnozaslonska kamera z zajemom + anotacijo

  1027→function CameraCapture({
  1028→  projectId,
  1029→  kategorija,
  1030→  onKategorijaChange,
  1031→  onClose,
  1032→  onSaved,
  1033→}: {
  1034→  projectId: string
  1035→  kategorija: 'PRED' | 'MED' | 'PO'

  1039→}) {

  1317→  const redraw = useCallback(() => {

  1360→  function onPointerDown(e: React.PointerEvent) {
