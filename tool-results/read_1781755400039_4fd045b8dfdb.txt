   246→  function calculateRailingClientSide() {
   247→    const L = parseFloat(effectiveTotalLength) * 1000
   248→    const W = parseFloat(slatWidth)
   249→    const G = parseFloat(maxGap)
   250→    const n = Math.ceil((L - G) / (G + W))
   251→    const actualGap = (L - n * W) / (n + 1)
   252→    const warnings: string[] = []
   253→
   254→    if (actualGap > 100) {
   255→      warnings.push('RAZMIK PRESEGA 100mm — Prepovedano za stanovanjske objekte!')
   256→    }
   257→    if (profileType === 'z-line') {
   258→      warnings.push('Z-line profil: Prekrivanje zagotavlja 100% vizualno zasebnost.')
   259→    }
   260→    if (actualGap < 10) {
   261→      warnings.push('Razmik zelo majhen (<10mm). Preverite dilatacijo WPC materiala.')
   262→    }
   263→
   264→    setRailingResult({
   265→      slatCount: n,
   266→      actualGapMm: Math.round(actualGap * 10) / 10,
   267→      totalSlatsLengthMm: n * W,
   268→      totalGapsLengthMm: Math.round((n + 1) * actualGap),
   269→      isCompliant: actualGap <= 100,
   270→      warnings,
   271→    })
   272→  }

   274→  function calculateAnchoringClientSide() {
   275→    const hc = parseInt(holeCount)
   276→    const depth = parseFloat(holeDepthMm)
   277→    const dia = parseFloat(holeDiameterMm)
   278→    const temp = parseFloat(temperature)
   279→    const warnings: string[] = []
   280→
   281→    const radiusMm = dia / 2
   282→    const holeVolumeMm3 = Math.PI * Math.pow(radiusMm, 2) * depth
   283→    const holeVolumeMl = holeVolumeMm3 / 1000
   284→    const resinPerHole = holeVolumeMl * 1.2
   285→    const totalResin = resinPerHole * hc
   286→
   287→    let curingTimeMin: number
   288→    if (temp >= 20) {
   289→      curingTimeMin = 30
   290→    } else if (temp >= 10) {
   291→      curingTimeMin = 60
   292→      warnings.push('Temperatura < 20°C: Podaljšan čas strjevanja. Počakajte vsaj 1 uro.')
   293→    } else if (temp >= 5) {
   294→      curingTimeMin = 120
   295→      warnings.push('Temperatura < 10°C: Zelo podaljšan čas strjevanja (2 uri). Uporabite zimsko formulo smole.')
   296→    } else {
   297→      curingTimeMin = 0
   298→      warnings.push('Temperatura < 5°C: Kemično sidranje NI priporočljivo!')
   299→    }
   300→
   301→    const cartridgeSize = anchorType === 'hilti-hit' ? 330 : 300
   302→    const cartridgesNeeded = Math.ceil(totalResin / cartridgeSize)
   303→
   304→    if (depth < 70) {
   305→      warnings.push('Globina vrtanja < 70mm. Priporočena minimalna globina za M12 sidro je 70mm.')
   306→    }
   307→
   308→    setAnchoringResult({
   309→      resinVolumeMl: Math.round(resinPerHole * 10) / 10,
   310→      totalResinMl: Math.round(totalResin * 10) / 10,
   311→      curingTimeMin,
   312→      cartridgesNeeded,
   313→      warnings,
   314→    })
   315→  }

   317→  function calculateWindClientSide() {
   318→    const h = parseFloat(heightAboveGround)
   319→    const terrainFactors: Record<string, number> = { I: 1.0, II: 0.91, III: 0.82, IV: 0.73 }
   320→    const kTerrain = terrainFactors[terrainCategory] || 0.91
   321→    const heightFactor = Math.pow(h / 10, 0.2)
   322→    const aeroFactors: Record<string, number> = { solid: 1.3, slatted: 0.8, 'z-line': 0.6 }
   323→    const cAero = aeroFactors[railingType] || 0.8
   324→
   325→    const ws = parseFloat(windSpeedMs)
   326→    const area = parseFloat(railingAreaM2)
   327→    const recommendations: string[] = []
   328→
   329→    const basePressure = 0.5 * 1.25 * Math.pow(ws, 2)
   330→    const designPressure = basePressure * kTerrain * heightFactor * cAero
   331→    const totalForce = designPressure * area
   332→    const railingLength = Math.sqrt(area)
   333→    const forcePerMeter = totalForce / railingLength
   334→
   335→    let riskLevel: 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL'
   336→    if (designPressure < 0.5) {
   337→      riskLevel = 'LOW'
   338→    } else if (designPressure < 1.0) {
   339→      riskLevel = 'MEDIUM'
   340→      recommendations.push('Preverite pritrdilne elemente. Uporabite A4 Inox vijake.')
   341→    } else if (designPressure < 1.5) {
   342→      riskLevel = 'HIGH'
   343→      recommendations.push('Visoka vetrna obremenitev! Uporabite kemično sidranje in dodatne stebre.')
   344→      recommendations.push('Priporočljivo: Z-line profil za zmanjšanje veternega upora.')
   345→    } else {
   346→      riskLevel = 'CRITICAL'
   347→      recommendations.push('KRITIČNA vetrna obremenitev! Potrebna statična analiza.')
   348→      recommendations.push('Obvezno: Kemično sidranje vseh stebrov, zmanjšan razmik med stebri.')
   349→    }
   350→
   351→    if (railingType === 'solid' && h > 20) {
   352→      recommendations.push('Polna ograja nad 20m: Tveganje harmoničnih vibracij. Vgradite dušilna tesnila.')
   353→    }
   354→
   355→    setWindResult({
   356→      windPressureKpa: Math.round(designPressure * 100) / 100,
   357→      totalForceKn: Math.round(totalForce * 100) / 100,
   358→      forcePerMeterNm: Math.round(forcePerMeter * 10) / 10,
   359→      riskLevel,
   360→      recommendations,
   361→    })
   362→  }
