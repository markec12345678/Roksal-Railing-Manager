   180→  function calculateRailingClientSide() {
   181→    const L = parseFloat(effectiveTotalLength) * 1000
   182→    const W = parseFloat(slatWidth)
   183→    const G = parseFloat(maxGap)
   184→    const n = Math.ceil((L - G) / (G + W))
   185→    const actualGap = (L - n * W) / (n + 1)
   186→    const warnings: string[] = []
   187→
   188→    if (actualGap > 100) {
   189→      warnings.push('RAZMIK PRESEGA 100mm — Prepovedano za stanovanjske objekte!')
   190→    }
   191→    if (profileType === 'z-line') {
   192→      warnings.push('Z-line profil: Prekrivanje zagotavlja 100% vizualno zasebnost.')
   193→    }
   194→    if (actualGap < 10) {
   195→      warnings.push('Razmik zelo majhen (<10mm). Preverite dilatacijo WPC materiala.')
   196→    }
   197→
   198→    setRailingResult({
   199→      slatCount: n,
   200→      actualGapMm: Math.round(actualGap * 10) / 10,
   201→      totalSlatsLengthMm: n * W,
   202→      totalGapsLengthMm: Math.round((n + 1) * actualGap),
   203→      isCompliant: actualGap <= 100,
   204→      warnings,
   205→    })

   208→  function calculateAnchoringClientSide() {
   209→    const hc = parseInt(holeCount)
   210→    const depth = parseFloat(holeDepthMm)
   211→    const dia = parseFloat(holeDiameterMm)
   212→    const temp = parseFloat(temperature)
   213→    const warnings: string[] = []
   214→
   215→    const radiusMm = dia / 2
   216→    const holeVolumeMm3 = Math.PI * Math.pow(radiusMm, 2) * depth
   217→    const holeVolumeMl = holeVolumeMm3 / 1000
   218→    const resinPerHole = holeVolumeMl * 1.2
   219→    const totalResin = resinPerHole * hc
   220→
   221→    let curingTimeMin: number
   222→    if (temp >= 20) {
   223→      curingTimeMin = 30
   224→    } else if (temp >= 10) {
   225→      curingTimeMin = 60
   226→      warnings.push('Temperatura < 20°C: Podaljšan čas strjevanja. Počakajte vsaj 1 uro.')
   227→    } else if (temp >= 5) {
   228→      curingTimeMin = 120
   229→      warnings.push('Temperatura < 10°C: Zelo podaljšan čas strjevanja (2 uri). Uporabite zimsko formulo smole.')
   230→    } else {
   231→      curingTimeMin = 0
   232→      warnings.push('Temperatura < 5°C: Kemično sidranje NI priporočljivo!')
   233→    }
   234→
   235→    const cartridgeSize = anchorType === 'hilti-hit' ? 330 : 300
   236→    const cartridgesNeeded = Math.ceil(totalResin / cartridgeSize)
   237→
   238→    if (depth < 70) {
   239→      warnings.push('Globina vrtanja < 70mm. Priporočena minimalna globina za M12 sidro je 70mm.')
   240→    }
   241→
   242→    setAnchoringResult({
   243→      resinVolumeMl: Math.round(resinPerHole * 10) / 10,
   244→      totalResinMl: Math.round(totalResin * 10) / 10,
   245→      curingTimeMin,
   246→      cartridgesNeeded,
   247→      warnings,
   248→    })
   249→  }

   251→  function calculateWindClientSide() {
   252→    const h = parseFloat(heightAboveGround)
   253→    const terrainFactors: Record<string, number> = { I: 1.0, II: 0.91, III: 0.82, IV: 0.73 }
   254→    const kTerrain = terrainFactors[terrainCategory] || 0.91
   255→    const heightFactor = Math.pow(h / 10, 0.2)
   256→    const aeroFactors: Record<string, number> = { solid: 1.3, slatted: 0.8, 'z-line': 0.6 }
   257→    const cAero = aeroFactors[railingType] || 0.8
   258→
   259→    const ws = parseFloat(windSpeedMs)
   260→    const area = parseFloat(railingAreaM2)
   261→    const recommendations: string[] = []
   262→
   263→    const basePressure = 0.5 * 1.25 * Math.pow(ws, 2)
   264→    const designPressure = basePressure * kTerrain * heightFactor * cAero
   265→    const totalForce = designPressure * area
   266→    const railingLength = Math.sqrt(area)
   267→    const forcePerMeter = totalForce / railingLength
   268→
   269→    let riskLevel: 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL'
   270→    if (designPressure < 0.5) {
   271→      riskLevel = 'LOW'
   272→    } else if (designPressure < 1.0) {
   273→      riskLevel = 'MEDIUM'
   274→      recommendations.push('Preverite pritrdilne elemente. Uporabite A4 Inox vijake.')
   275→    } else if (designPressure < 1.5) {
   276→      riskLevel = 'HIGH'
   277→      recommendations.push('Visoka vetrna obremenitev! Uporabite kemično sidranje in dodatne stebre.')
   278→      recommendations.push('Priporočljivo: Z-line profil za zmanjšanje veternega upora.')
   279→    } else {
   280→      riskLevel = 'CRITICAL'
   281→      recommendations.push('KRITIČNA vetrna obremenitev! Potrebna statična analiza.')
   282→      recommendations.push('Obvezno: Kemično sidranje vseh stebrov, zmanjšan razmik med stebri.')
   283→    }
   284→
   285→    if (railingType === 'solid' && h > 20) {
   286→      recommendations.push('Polna ograja nad 20m: Tveganje harmoničnih vibracij. Vgradite dušilna tesnila.')
   287→    }
   288→
   289→    setWindResult({
   290→      windPressureKpa: Math.round(designPressure * 100) / 100,
   291→      totalForceKn: Math.round(totalForce * 100) / 100,
   292→      forcePerMeterNm: Math.round(forcePerMeter * 10) / 10,
   293→      riskLevel,
   294→      recommendations,
   295→    })
   296→  }
