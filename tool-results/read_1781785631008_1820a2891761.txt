    86→    86→// Komprimira datoteko v base64 JPEG (max 1280px širina, kvaliteta 0.75)
    87→    87→function compressImageFile(file: File, maxSize = 1280, quality = 0.75): Promise<string> {
    88→    88→  return new Promise((resolve, reject) => {
    89→    89→    const reader = new FileReader()
    90→    90→    reader.onload = () => {
    91→    91→      const img = new Image()
    92→    92→      img.onload = () => {
    93→    93→        const scale = Math.min(1, maxSize / Math.max(1, img.naturalWidth))
    94→    94→        const canvas = document.createElement('canvas')
    95→    95→        canvas.width = Math.max(1, Math.round(img.naturalWidth * scale))
    96→    96→        canvas.height = Math.max(1, Math.round(img.naturalHeight * scale))
    97→    97→        const ctx = canvas.getContext('2d')
    98→    98→        if (!ctx) return reject(new Error('Canvas ni podprt'))
    99→    99→        ctx.drawImage(img, 0, 0, canvas.width, canvas.height)
   100→   100→        resolve(canvas.toDataURL('image/jpeg', quality))
   101→   101→      }
   102→   102→      img.onerror = () => reject(new Error('Napaka pri nalaganju slike'))
   103→   103→      img.src = reader.result as string
   104→   104→    }
   105→   105→    reader.onerror = () => reject(new Error('Napaka pri branju datoteke'))
   106→   106→    reader.readAsDataURL(file)
   107→   107→  })
   108→   108→}

   111→   111→function estimateBytes(base64: string): number {
   112→   112→  const idx = base64.indexOf(',')
   113→   113→  const b64 = idx >= 0 ? base64.slice(idx + 1) : base64
   114→   114→  return Math.floor(b64.length * 0.75)
   115→   115→}

   117→   117→function formatBytes(bytes: number): string {
   118→   118→  if (bytes < 1024) return `${bytes} B`
   119→   119→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} KB`
   120→   120→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   121→   121→}

   123→   123→function formatRelativeTime(iso: string): string {
   124→   124→  const date = new Date(iso)
   125→   125→  const diff = Date.now() - date.getTime()
   126→   126→  const sec = Math.floor(diff / 1000)
   127→   127→  if (sec < 60) return 'pravkar'
   128→   128→  const min = Math.floor(sec / 60)
   129→   129→  if (min < 60) return `pred ${min} min`
   130→   130→  const h = Math.floor(min / 60)
   131→   131→  if (h < 24) return `pred ${h} h`
   132→   132→  const d = Math.floor(h / 24)
   133→   133→  if (d < 7) return `pred ${d} d`
   134→   134→  return date.toLocaleDateString('sl-SI')
   135→   135→}

   142→   142→// Nariše puščično glavo
   143→   143→function drawArrowHead(
   144→   144→  ctx: CanvasRenderingContext2D,
   145→   145→  x1: number, y1: number,
   146→   146→  x2: number, y2: number,
   147→   147→  color: string, width: number
   148→   148→) {
   149→   149→  const angle = Math.atan2(y2 - y1, x2 - x1)
   150→   150→  const headLen = Math.max(10, width * 3)
   151→   151→  ctx.strokeStyle = color
   152→   152→  ctx.fillStyle = color
   153→   153→  ctx.lineWidth = width
   154→   154→  ctx.beginPath()
   155→   155→  ctx.moveTo(x2, y2)
   156→   156→  ctx.lineTo(x2 - headLen * Math.cos(angle - Math.PI / 6), y2 - headLen * Math.sin(angle - Math.PI / 6))
   157→   157→  ctx.moveTo(x2, y2)
   158→   158→  ctx.lineTo(x2 - headLen * Math.cos(angle + Math.PI / 6), y2 - headLen * Math.sin(angle + Math.PI / 6))
   159→   159→  ctx.stroke()
   160→   160→}

   162→   162→// Nariše posamezno anotacijo na kontekst
   163→   163→function drawAnnotation(ctx: CanvasRenderingContext2D, ann: Annotation) {
   164→   164→  ctx.strokeStyle = ann.color
   165→   165→  ctx.fillStyle = ann.color
   166→   166→  ctx.lineWidth = Math.max(0.5, ann.width)
   167→   167→  ctx.lineCap = 'round'
   168→   168→  ctx.lineJoin = 'round'
   169→   169→
   170→   170→  if (ann.type === 'pen' && ann.points.length > 0) {
   171→   171→    ctx.beginPath()
   172→   172→    ctx.moveTo(ann.points[0].x, ann.points[0].y)
   173→   173→    for (let i = 1; i < ann.points.length; i++) {
   174→   174→      ctx.lineTo(ann.points[i].x, ann.points[i].y)
   175→   175→    }
   176→   176→    ctx.stroke()
   177→   177→    return
   178→   178→  }
   179→   179→
   180→   180→  if (ann.type === 'text' && ann.text && ann.points.length >= 1) {
   181→   181→    const p = ann.points[0]
   182→   182→    const fs = ann.fontSize ?? 18
   183→   183→    ctx.font = `bold ${fs}px sans-serif`
   184→   184→    ctx.textAlign = 'left'
   185→   185→    ctx.textBaseline = 'top'
   186→   186→    // Beli rob za vidnost
   187→   187→    ctx.lineWidth = Math.max(2, fs * 0.18)
   188→   188→    ctx.strokeStyle = ann.color === '#ffffff' ? '#1d2b3e' : '#ffffff'
   189→   189→    ctx.strokeText(ann.text, p.x, p.y)
   190→   190→    ctx.fillStyle = ann.color
   191→   191→    ctx.fillText(ann.text, p.x, p.y)
   192→   192→    return
   193→   193→  }
   194→   194→
   195→   195→  if (ann.points.length < 2) return
   196→   196→
   197→   197→  const [a, b] = ann.points
   198→   198→  if (ann.type === 'arrow') {
   199→   199→    ctx.beginPath()
   200→   200→    ctx.moveTo(a.x, a.y)
   201→   201→    ctx.lineTo(b.x, b.y)
   202→   202→    ctx.stroke()
   203→   203→    drawArrowHead(ctx, a.x, a.y, b.x, b.y, ann.color, ann.width)
   204→   204→  } else if (ann.type === 'line') {
   205→   205→    ctx.beginPath()
   206→   206→    ctx.moveTo(a.x, a.y)
   207→   207→    ctx.lineTo(b.x, b.y)
   208→   208→    ctx.stroke()
   209→   209→  } else if (ann.type === 'measure') {
   210→   210→    ctx.beginPath()
   211→   211→    ctx.moveTo(a.x, a.y)
   212→   212→    ctx.lineTo(b.x, b.y)
   213→   213→    ctx.stroke()
   214→   214→    drawArrowHead(ctx, a.x, a.y, b.x, b.y, ann.color, ann.width)
   215→   215→    drawArrowHead(ctx, b.x, b.y, a.x, a.y, ann.color, ann.width)
   216→   216→    if (ann.label) {
   217→   217→      const mx = (a.x + b.x) / 2
   218→   218→      const my = (a.y + b.y) / 2
   219→   219→      const fs = ann.fontSize ?? 14
   220→   220→      ctx.font = `bold ${fs}px sans-serif`
   221→   221→      const w = ctx.measureText(ann.label).width + 12
   222→   222→      ctx.fillStyle = ann.color
   223→   223→      ctx.fillRect(mx - w / 2, my - fs / 2 - 4, w, fs + 8)
   224→   224→      ctx.fillStyle = ann.color === '#ffffff' ? '#1d2b3e' : '#ffffff'
   225→   225→      ctx.textAlign = 'center'
   226→   226→      ctx.textBaseline = 'middle'
   227→   227→      ctx.fillText(ann.label, mx, my)
   228→   228→    }
   229→   229→  } else if (ann.type === 'rect') {
   230→   230→    const x = Math.min(a.x, b.x)
   231→   231→    const y = Math.min(a.y, b.y)
   232→   232→    const w = Math.abs(b.x - a.x)
   233→   233→    const h = Math.abs(b.y - a.y)
   234→   234→    ctx.strokeRect(x, y, w, h)
   235→   235→  } else if (ann.type === 'circle') {
   236→   236→    const cx = a.x
   237→   237→    const cy = a.y
   238→   238→    const r = Math.max(1, Math.hypot(b.x - a.x, b.y - a.y))
   239→   239→    ctx.beginPath()
   240→   240→    ctx.arc(cx, cy, r, 0, Math.PI * 2)
   241→   241→    ctx.stroke()
   242→   242→  }
   243→   243→}

   431→   431→  async function handleBatchFiles(files: FileList) {
   432→   432→    if (!projectId || files.length === 0) return

   434→   434→    setBatchProgress({ current: 0, total: files.length })
   435→   435→

   440→   440→      await new Promise<void>((resolve) => {
   441→   441→        navigator.geolocation.getCurrentPosition(
   442→   442→          (pos) => { lat = pos.coords.latitude; lon = pos.coords.longitude; resolve() },
   443→   443→          () => resolve(),
   444→   444→          { enableHighAccuracy: true, timeout: 3000 }
   445→   445→        )
   446→   446→      })
   447→   447→    }
   448→   448→
   449→   449→    let success = 0

   451→   451→      setBatchProgress({ current: i, total: files.length })

   453→   453→        const dataUrl = await compressImageFile(files[i])

   461→   461→            latitude: lat,
   462→   462→            longitude: lon,
   463→   463→          }),
   464→   464→        })
   465→   465→        if (res.ok) success++
   466→   466→      } catch (e) {
   467→   467→        console.error('Batch upload error:', e)
   468→   468→      }

   472→   472→    setBatchProgress(null)

  1025→  1025→// CAMERA CAPTURE — polnozaslonska kamera z zajemom + anotacijo
  1026→  1026→// ============================================================
  1027→  1027→function CameraCapture({
  1028→  1028→  projectId,
  1029→  1029→  kategorija,
  1030→  1030→  onKategorijaChange,
  1031→  1031→  onClose,
  1032→  1032→  onSaved,
  1033→  1033→}: {
  1034→  1034→  projectId: string
  1035→  1035→  kategorija: 'PRED' | 'MED' | 'PO'
  1036→  1036→  onKategorijaChange: (k: 'PRED' | 'MED' | 'PO') => void
  1037→  1037→  onClose: () => void
  1038→  1038→  onSaved: () => void
  1039→  1039→}) {
  1040→  1040→  const videoRef = useRef<HTMLVideoElement>(null)
  1041→  1041→  const streamRef = useRef<MediaStream | null>(null)
  1042→  1042→  const [status, setStatus] = useState<'starting' | 'ready' | 'error' | 'captured'>('starting')
  1043→  1043→  const [errorMsg, setErrorMsg] = useState('')
  1044→  1044→  const [capturedData, setCapturedData] = useState<string | null>(null)
  1045→  1045→  const [opomba, setOpomba] = useState('')
  1046→  1046→  const [gps, setGps] = useState<{ lat: number; lon: number } | null>(null)
  1047→  1047→  const [saving, setSaving] = useState(false)
  1048→  1048→  const [annotateMode, setAnnotateMode] = useState(false)
  1049→  1049→  const { toast } = useToast()
  1050→  1050→
  1051→  1051→  useEffect(() => {
  1052→  1052→    let mounted = true
  1053→  1053→    async function start() {
  1054→  1054→      try {
  1055→  1055→        const stream = await navigator.mediaDevices.getUserMedia({
  1056→  1056→          video: { facingMode: 'environment', width: { ideal: 1920 }, height: { ideal: 1080 } },
  1057→  1057→          audio: false,
  1058→  1058→        })
  1059→  1059→        if (!mounted) {
  1060→  1060→          stream.getTracks().forEach((t) => t.stop())
  1061→  1061→          return
  1062→  1062→        }
  1063→  1063→        streamRef.current = stream
  1064→  1064→        if (videoRef.current) {
  1065→  1065→          videoRef.current.srcObject = stream
  1066→  1066→          await videoRef.current.play()
  1067→  1067→        }
  1068→  1068→        setStatus('ready')
  1069→  1069→      } catch (e) {
  1070→  1070→        setStatus('error')
  1071→  1071→        setErrorMsg(e instanceof Error ? e.message : 'Kamera ni na voljo')
  1072→  1072→      }
  1073→  1073→    }
  1074→  1074→    start()
  1075→  1075→
  1076→  1076→    if (navigator.geolocation) {
  1077→  1077→      navigator.geolocation.getCurrentPosition(
  1078→  1078→        (pos) => setGps({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
  1079→  1079→        () => {},
  1080→  1080→        { enableHighAccuracy: true, timeout: 5000 }
  1081→  1081→      )
  1082→  1082→    }
  1083→  1083→
  1084→  1084→    return () => {
  1085→  1085→      mounted = false
  1086→  1086→      if (streamRef.current) {
  1087→  1087→        streamRef.current.getTracks().forEach((t) => t.stop())
  1088→  1088→        streamRef.current = null
  1089→  1089→      }
  1090→  1090→    }
  1091→  1091→  }, [])
  1092→  1092→
  1093→  1093→  function handleCapture() {
  1094→  1094→    if (!videoRef.current) return
  1095→  1095→    const video = videoRef.current
  1096→  1096→    const canvas = document.createElement('canvas')
  1097→  1097→    const scale = Math.min(1, 1280 / Math.max(1, video.videoWidth))
  1098→  1098→    canvas.width = Math.max(1, Math.round(video.videoWidth * scale))
  1099→  1099→    canvas.height = Math.max(1, Math.round(video.videoHeight * scale))
  1100→  1100→    const ctx = canvas.getContext('2d')
  1101→  1101→    if (!ctx) return
  1102→  1102→    ctx.drawImage(video, 0, 0, canvas.width, canvas.height)
  1103→  1103→    const data = canvas.toDataURL('image/jpeg', 0.75)
  1104→  1104→    setCapturedData(data)
  1105→  1105→    setStatus('captured')
  1106→  1106→    if (streamRef.current) {
  1107→  1107→      streamRef.current.getTracks().forEach((t) => t.stop())
  1108→  1108→      streamRef.current = null
  1109→  1109→    }
  1110→  1110→  }
  1111→  1111→
  1112→  1112→  function handleRetake() {
  1113→  1113→    setCapturedData(null)
  1114→  1114→    setStatus('starting')
  1115→  1115→    ;(async () => {
  1116→  1116→      try {
  1117→  1117→        const stream = await navigator.mediaDevices.getUserMedia({
  1118→  1118→          video: { facingMode: 'environment' },
  1119→  1119→          audio: false,
  1120→  1120→        })
  1121→  1121→        streamRef.current = stream
  1122→  1122→        if (videoRef.current) {
  1123→  1123→          videoRef.current.srcObject = stream
  1124→  1124→          await videoRef.current.play()
  1125→  1125→        }
  1126→  1126→        setStatus('ready')
  1127→  1127→      } catch {
  1128→  1128→        setStatus('error')
  1129→  1129→      }
  1130→  1130→    })()
  1131→  1131→  }
  1132→  1132→
  1133→  1133→  async function handleSave() {
  1134→  1134→    if (!capturedData) return
  1135→  1135→    setSaving(true)
  1136→  1136→    try {
  1137→  1137→      const res = await fetch('/api/photos', {
  1138→  1138→        method: 'POST',
  1139→  1139→        headers: { 'Content-Type': 'application/json' },
  1140→  1140→        body: JSON.stringify({
  1141→  1141→          projectId,
  1142→  1142→          kategorija,
  1143→  1143→          imageData: capturedData,
  1144→  1144→          opomba: opomba || null,
  1145→  1145→          latitude: gps?.lat ?? null,
  1146→  1146→          longitude: gps?.lon ?? null,
  1147→  1147→        }),
  1148→  1148→      })
  1149→  1149→      if (res.ok) {
  1150→  1150→        toast({
  1151→  1151→          title: 'Slika shranjena',
  1152→  1152→          description: `${KATEGORIJE.find((k) => k.id === kategorija)?.label}${gps ? ' · GPS zabeležen' : ''}`,
  1153→  1153→        })
  1154→  1154→        onSaved()
  1155→  1155→      } else {
  1156→  1156→        toast({ title: 'Napaka pri shranjevanju', variant: 'destructive' })
  1157→  1157→      }
  1158→  1158→    } catch {
  1159→  1159→      toast({ title: 'Omrežna napaka', variant: 'destructive' })
  1160→  1160→    } finally {
  1161→  1161→      setSaving(false)
  1162→  1162→    }
  1163→  1163→  }
  1164→  1164→
  1165→  1165→  function handleClose() {
  1166→  1166→    if (streamRef.current) {
  1167→  1167→      streamRef.current.getTracks().forEach((t) => t.stop())
  1168→  1168→    }

  1170→  1170→  }
  1171→  1171→
  1172→  1172→  // Annotation editor nad zajeto sliko
  1173→  1173→  if (annotateMode && capturedData) {
  1174→  1174→    return (
  1175→  1175→      <AnnotationEditor
  1176→  1176→        imageData={capturedData}
  1177→  1177→        onSave={(newData) => {

  1185→  1185→  }
  1186→  1186→
  1187→  1187→  return (
  1188→  1188→    <div className="fixed inset-0 z-[60] flex flex-col bg-black">

  1191→  1191→        <button type="button" onClick={handleClose} className="rounded-full p-1.5 hover:bg-white/10">
  1192→  1192→          <X className="h-5 w-5" />
  1193→  1193→        </button>
  1194→  1194→        <span className="text-sm font-medium">Slikanje — {KATEGORIJE.find((k) => k.id === kategorija)?.label}</span>
