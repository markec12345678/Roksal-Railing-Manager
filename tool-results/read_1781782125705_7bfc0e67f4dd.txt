   156→   156→function dist(a: XY, b: XY): number {
   157→   157→  const dx = b.x - a.x
   158→   158→  const dy = b.y - a.y
   159→   159→  return Math.sqrt(dx * dx + dy * dy)
   160→   160→}
   161→   161→
   162→   162→function formatDistance(mm: number): string {
   163→   163→  if (!Number.isFinite(mm) || mm <= 0) return '—'
   164→   164→  if (mm >= 1000) return `${(mm / 1000).toFixed(2)} m`
   165→   165→  if (mm >= 100) return `${(mm / 10).toFixed(1)} cm`
   166→   166→  return `${Math.round(mm)} mm`
   167→   167→}
   168→   168→
   169→   169→function findNearestPoint(points: Tocka[], p: XY): number {
   170→   170→  let best = -1
   171→   171→  let bestD = HIT_RADIUS
   172→   172→  for (let i = 0; i < points.length; i++) {
   173→   173→    const d = dist(points[i], p)
   174→   174→    if (d < bestD) {
   175→   175→      bestD = d
   176→   176→      best = i
   177→   177→    }
   178→   178→  }
   179→   179→  return best
   180→   180→}

   190→   190→function drawAnchorPoint(
   191→   191→  ctx: CanvasRenderingContext2D,
   192→   192→  p: Tocka,
   193→   193→  index: number,
   194→   194→  total: number,
   195→   195→): void {
   196→   196→  const isEnd = total <= 2 || index === 0 || index === total - 1
   197→   197→  const r = isEnd ? 14 : 10
   198→   198→  // Outer halo for end posts
   199→   199→  if (isEnd) {
   200→   200→    ctx.beginPath()
   201→   201→    ctx.arc(p.x, p.y, r + 4, 0, Math.PI * 2)
   202→   202→    ctx.fillStyle = 'rgba(29, 43, 62, 0.18)'
   203→   203→    ctx.fill()
   204→   204→  }
   205→   205→  // Filled circle
   206→   206→  ctx.beginPath()
   207→   207→  ctx.arc(p.x, p.y, r, 0, Math.PI * 2)
   208→   208→  ctx.fillStyle = isEnd ? NAVY : AMBER
   209→   209→  ctx.fill()
   210→   210→  ctx.strokeStyle = '#ffffff'
   211→   211→  ctx.lineWidth = 2.5
   212→   212→  ctx.stroke()
   213→   213→  // Number label
   214→   214→  ctx.fillStyle = '#ffffff'
   215→   215→  ctx.font = `bold ${isEnd ? 13 : 11}px ui-sans-serif, system-ui, sans-serif`
   216→   216→  ctx.textAlign = 'center'
   217→   217→  ctx.textBaseline = 'middle'
   218→   218→  ctx.fillText(String(index + 1), p.x, p.y)
   219→   219→}

   221→   221→function drawInfill(
   222→   222→  ctx: CanvasRenderingContext2D,
   223→   223→  a: XY, // bottom-left (anchor i)
   224→   224→  b: XY, // bottom-right (anchor i+1)
   225→   225→  ta: XY, // top-left (post top i)
   226→   226→  tb: XY, // top-right (post top i+1)
   227→   227→  profil: Profil,
   228→   228→  slatSpacingPx: number,
   229→   229→): void {
   230→   230→  const kat = profil.kategorija.toLowerCase()
   231→   231→  const isWpcVodoravno = kat.includes('vodoravno')
   232→   232→  const isWpcPokoncno =
   233→   233→    kat.includes('pokončno') || kat.includes('pokoncno')
   234→   234→  const isInox = kat.includes('inox')
   235→   235→  const isSteklo = kat.includes('steklo')
   236→   236→  const isAluKlasicno =
   237→   237→    kat.includes('klasično') || kat.includes('klasicno')
   238→   238→
   239→   239→  // --- Steklo: translucent panel ---
   240→   240→  if (isSteklo) {
   241→   241→    ctx.save()
   242→   242→    ctx.fillStyle = 'rgba(186, 230, 253, 0.35)'
   243→   243→    ctx.strokeStyle = 'rgba(125, 211, 252, 0.7)'
   244→   244→    ctx.lineWidth = 2
   245→   245→    ctx.beginPath()
   246→   246→    ctx.moveTo(a.x, a.y)
   247→   247→    ctx.lineTo(b.x, b.y)
   248→   248→    ctx.lineTo(tb.x, tb.y)
   249→   249→    ctx.lineTo(ta.x, ta.y)
   250→   250→    ctx.closePath()
   251→   251→    ctx.fill()
   252→   252→    ctx.stroke()
   253→   253→    // Subtle highlight
   254→   254→    ctx.strokeStyle = 'rgba(255, 255, 255, 0.45)'
   255→   255→    ctx.lineWidth = 1
   256→   256→    ctx.beginPath()
   257→   257→    ctx.moveTo(ta.x + 4, ta.y + 4)
   258→   258→    ctx.lineTo(tb.x - 4, tb.y + 4)
   259→   259→    ctx.stroke()
   260→   260→    ctx.restore()
   261→   261→    return
   262→   262→  }
   263→   263→
   264→   264→  // --- WPC vodoravno: horizontal slats from bottom up ---
   265→   265→  if (isWpcVodoravno) {
   266→   266→    const segHeight = Math.max(
   267→   267→      Math.abs(ta.y - a.y),
   268→   268→      Math.abs(tb.y - b.y),
   269→   269→    )
   270→   270→    const numSlats = Math.max(1, Math.floor(segHeight / slatSpacingPx))
   271→   271→    const actualSpacing = segHeight / numSlats
   272→   272→    ctx.strokeStyle = '#8b5a2b'
   273→   273→    ctx.lineWidth = 5
   274→   274→    ctx.lineCap = 'round'
   275→   275→    for (let s = 1; s <= numSlats; s++) {
   276→   276→      const offset = s * actualSpacing
   277→   277→      const ly = a.y - offset
   278→   278→      const ry = b.y - offset
   279→   279→      // Stop if both endpoints rise above the top rail
   280→   280→      if (ly < ta.y && ry < tb.y) break
   281→   281→      const clampedLy = Math.max(ly, ta.y)
   282→   282→      const clampedRy = Math.max(ry, tb.y)
   283→   283→      ctx.beginPath()
   284→   284→      ctx.moveTo(a.x, clampedLy)
   285→   285→      ctx.lineTo(b.x, clampedRy)
   286→   286→      ctx.stroke()
   287→   287→    }
   288→   288→    return
   289→   289→  }
   290→   290→
   291→   291→  // --- WPC pokončno / Inox: vertical balusters ---
   292→   292→  if (isWpcPokoncno || isInox) {
   293→   293→    const segLen = dist(a, b)
   294→   294→    const numBal = Math.max(1, Math.floor(segLen / slatSpacingPx))
   295→   295→    ctx.strokeStyle = isInox ? '#c0c4cc' : '#8b5a2b'
   296→   296→    ctx.lineWidth = isInox ? 2.5 : 3.5
   297→   297→    ctx.lineCap = 'round'
   298→   298→    for (let s = 1; s < numBal; s++) {
   299→   299→      const t = s / numBal
   300→   300→      const bx = a.x + (b.x - a.x) * t
   301→   301→      const by = a.y + (b.y - a.y) * t
   302→   302→      const tx = ta.x + (tb.x - ta.x) * t
   303→   303→      const ty = ta.y + (tb.y - ta.y) * t
   304→   304→      ctx.beginPath()
   305→   305→      ctx.moveTo(bx, by)
   306→   306→      ctx.lineTo(tx, ty)
   307→   307→      ctx.stroke()
   308→   308→    }
   309→   309→    return
   310→   310→  }
   311→   311→
   312→   312→  // --- Alu klasično: bottom rail + vertical pickets ---
   313→   313→  if (isAluKlasicno) {
   314→   314→    // Bottom rail
   315→   315→    ctx.strokeStyle = NAVY
   316→   316→    ctx.lineWidth = 5
   317→   317→    ctx.lineCap = 'round'
   318→   318→    ctx.beginPath()
   319→   319→    ctx.moveTo(a.x, a.y)
   320→   320→    ctx.lineTo(b.x, b.y)
   321→   321→    ctx.stroke()
   322→   322→    // Vertical pickets
   323→   323→    const segLen = dist(a, b)
   324→   324→    const numPick = Math.max(1, Math.floor(segLen / slatSpacingPx))
   325→   325→    ctx.strokeStyle = '#6b7280'
   326→   326→    ctx.lineWidth = 2
   327→   327→    for (let s = 1; s < numPick; s++) {
   328→   328→      const t = s / numPick
   329→   329→      const bx = a.x + (b.x - a.x) * t
   330→   330→      const by = a.y + (b.y - a.y) * t
   331→   331→      const tx = ta.x + (tb.x - ta.x) * t
   332→   332→      const ty = ta.y + (tb.y - ta.y) * t
   333→   333→      ctx.beginPath()
   334→   334→      ctx.moveTo(bx, by)
   335→   335→      ctx.lineTo(tx, ty)
   336→   336→      ctx.stroke()
   337→   337→    }
   338→   338→    return
   339→   339→  }
   340→   340→
   341→   341→  // Fallback: no infill
   342→   342→}
   343→   343→
   344→   344→function drawRailing(
   345→   345→  ctx: CanvasRenderingContext2D,
   346→   346→  tocke: Tocka[],
   347→   347→  profil: Profil,
   348→   348→  postHeightPx: number,
   349→   349→  slatSpacingPx: number,
   350→   350→): void {
   351→   351→  if (tocke.length < 2) return
   352→   352→  const cappedHeight = Math.min(postHeightPx, POST_HEIGHT_CAP_PX)
   353→   353→  const tops: XY[] = tocke.map((p) => ({
   354→   354→    x: p.x,
   355→   355→    y: p.y - cappedHeight,
   356→   356→  }))
   357→   357→
   358→   358→  // 1. Infill between consecutive posts (drawn first so posts/rails overlay)
   359→   359→  for (let i = 0; i < tocke.length - 1; i++) {
   360→   360→    drawInfill(ctx, tocke[i], tocke[i + 1], tops[i], tops[i + 1], profil, slatSpacingPx)
   361→   361→  }
   362→   362→
   363→   363→  // 2. Vertical posts at each anchor (navy)
   364→   364→  ctx.strokeStyle = NAVY
   365→   365→  ctx.lineWidth = 6
   366→   366→  ctx.lineCap = 'round'
   367→   367→  for (let i = 0; i < tocke.length; i++) {
   368→   368→    ctx.beginPath()
   369→   369→    ctx.moveTo(tocke[i].x, tocke[i].y)
   370→   370→    ctx.lineTo(tops[i].x, tops[i].y)
   371→   371→    ctx.stroke()
   372→   372→  }
   373→   373→
   374→   374→  // 3. Top rail — thick navy line connecting tops of consecutive posts
   375→   375→  ctx.strokeStyle = NAVY
   376→   376→  ctx.lineWidth = 8
   377→   377→  ctx.lineCap = 'round'
   378→   378→  ctx.lineJoin = 'round'
   379→   379→  ctx.beginPath()
   380→   380→  ctx.moveTo(tops[0].x, tops[0].y)
   381→   381→  for (let i = 1; i < tops.length; i++) {
   382→   382→    ctx.lineTo(tops[i].x, tops[i].y)
   383→   383→  }
   384→   384→  ctx.stroke()
   385→   385→}
   386→   386→
   387→   387→function drawMeasurement(ctx: CanvasRenderingContext2D, m: Meritev): void {
   388→   388→  // Dashed line
   389→   389→  ctx.save()
   390→   390→  ctx.strokeStyle = GREEN
   391→   391→  ctx.lineWidth = 2.5
   392→   392→  ctx.setLineDash([8, 4])
   393→   393→  ctx.beginPath()
   394→   394→  ctx.moveTo(m.a.x, m.a.y)
   395→   395→  ctx.lineTo(m.b.x, m.b.y)
   396→   396→  ctx.stroke()
   397→   397→  ctx.setLineDash([])
   398→   398→  // End markers
   399→   399→  for (const pt of [m.a, m.b]) {
   400→   400→    ctx.beginPath()
   401→   401→    ctx.arc(pt.x, pt.y, 6, 0, Math.PI * 2)
   402→   402→    ctx.fillStyle = GREEN
   403→   403→    ctx.fill()
   404→   404→    ctx.strokeStyle = '#ffffff'
   405→   405→    ctx.lineWidth = 2
   406→   406→    ctx.stroke()
   407→   407→  }
   408→   408→  // Label badge at midpoint
   409→   409→  const mx = (m.a.x + m.b.x) / 2
   410→   410→  const my = (m.a.y + m.b.y) / 2
   411→   411→  const label = `${m.oznaka}: ${formatDistance(m.dolzinaMm)}`
   412→   412→  ctx.font = 'bold 12px ui-sans-serif, system-ui, sans-serif'
   413→   413→  const metrics = ctx.measureText(label)
   414→   414→  const pad = 6
   415→   415→  const w = metrics.width + pad * 2
   416→   416→  const h = 20
   417→   417→  // Background pill
   418→   418→  ctx.fillStyle = 'rgba(29, 43, 62, 0.92)'
   419→   419→  if (typeof ctx.roundRect === 'function') {
   420→   420→    ctx.beginPath()
   421→   421→    ctx.roundRect(mx - w / 2, my - h / 2, w, h, 4)
   422→   422→    ctx.fill()
   423→   423→  } else {
   424→   424→    ctx.fillRect(mx - w / 2, my - h / 2, w, h)
   425→   425→  }
   426→   426→  // Text
   427→   427→  ctx.fillStyle = '#ffffff'
   428→   428→  ctx.textAlign = 'center'
   429→   429→  ctx.textBaseline = 'middle'
   430→   430→  ctx.fillText(label, mx, my)
   431→   431→  ctx.restore()
   432→   432→}
