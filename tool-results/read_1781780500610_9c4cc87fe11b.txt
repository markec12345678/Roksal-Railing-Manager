   108→   108→  const d = new Date(dateStr)

   115→   115→function formatDateNice(dateStr: string): string {
   116→   116→  const d = new Date(dateStr)
   117→   117→  const now = new Date()
   118→   118→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   119→   119→  const date = new Date(d.getFullYear(), d.getMonth(), d.getDate())
   120→   120→  const diffDays = Math.round((today.getTime() - date.getTime()) / 86400000)
   121→   121→
   122→   122→  if (diffDays === 0) return 'danes'
   123→   123→  if (diffDays === 1) return 'včeraj'
   124→   124→  if (diffDays === -1) return 'jutri'
   125→   125→  if (diffDays === -2) return 'pjutri'
   126→   126→  if (diffDays > 1 && diffDays <= 7) return `pred ${diffDays} dnevi`
   127→   127→  if (diffDays > 7 && diffDays <= 30) return `pred ${Math.floor(diffDays / 7)} tedni`
   128→   128→  if (diffDays < -2 && diffDays >= -7) return `čez ${Math.abs(diffDays)} dni`
   129→   129→  return formatDate(dateStr)
   130→   130→}

   132→   132→function formatRelativeTime(dateStr: string): string {
   133→   133→  const now = new Date()
   134→   134→  const date = new Date(dateStr)
   135→   135→  const diffMs = now.getTime() - date.getTime()
   136→   136→  const diffMin = Math.floor(diffMs / 60000)
   137→   137→  const diffH = Math.floor(diffMs / 3600000)
   138→   138→  const diffD = Math.floor(diffMs / 86400000)
   139→   139→
   140→   140→  if (diffMin < 1) return 'Pravkar'
   141→   141→  if (diffMin < 60) return `pred ${diffMin} min`
   142→   142→  if (diffH < 24) return `pred ${diffH} ur`
   143→   143→  if (diffD < 7) return `pred ${diffD} dni`
   144→   144→  return formatDate(dateStr)
   145→   145→}

   181→   181→function MiniRailingDiagram({ dolzina, visina }: { dolzina: number; visina: number }) {
   182→   182→  const maxDim = Math.max(dolzina, visina)
   183→   183→  const heightPct = Math.min((visina / maxDim) * 30, 30)
   184→   184→  const n = Math.ceil((dolzina - 100) / (100 + 80))
   185→   185→  const actualGap = (dolzina - n * 80) / (n + 1)
   186→   186→  const numSlats = Math.min(n, 12)
   187→   187→  const gapPct = actualGap / dolzina * 100
   188→   188→
   189→   189→  return (
   190→   190→    <div className="relative rounded-md border border-roksal-navy/15 bg-gradient-to-b from-roksal-navy/3 to-roksal-navy/6 p-2" style={{ minHeight: `${Math.max(heightPct, 14)}px` }}>
   191→   191→      <div className="flex items-end h-full gap-0" style={{ height: `${Math.max(heightPct, 14)}px` }}>
   192→   192→        <div className="w-[3px] h-full bg-roksal-navy rounded-full" />
   193→   193→        <div className="flex-1 flex items-end h-full gap-0">
   194→   194→          <div className="h-full bg-transparent" style={{ width: `${gapPct}%` }} />

   197→   197→              <div className="h-[85%] bg-roksal-navy/70 rounded-[1px]" style={{ width: `${(80 / dolzina) * 100}%`, minWidth: '1px' }} />
