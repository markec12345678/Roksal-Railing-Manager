   217→function parseArMetadata(raw: string | null | undefined): ArMetadata {
   218→  if (!raw) return {}
   219→  try {
   220→    return JSON.parse(raw) as ArMetadata
   221→  } catch {
   222→    return {}
   223→  }
   224→}

   226→function parseGPS(gpsStr: string | null): { lat: number; lng: number } | null {
   227→  if (!gpsStr) return null
   228→  try {
   229→    return JSON.parse(gpsStr)
   230→  } catch {
   231→    return null
   232→  }
   233→}

   235→function formatDimension(mm: number): string {
   236→  if (mm >= 1000) return `${(mm / 1000).toFixed(2)}m`
   237→  return `${mm}mm`
   238→}
   239→
   240→function formatM2(mm2: number): string {
   241→  return `${(mm2 / 1_000_000).toFixed(2)}m²`
   242→}

   548→  // Hitri izračun razmikov
   549→  function getQuickSpacing(dolzinaMm: number, visinaMm: number, slatWidth = 80, maxGap = 100) {
   550→    const n = Math.ceil((dolzinaMm - maxGap) / (maxGap + slatWidth))
   551→    const actualGap = (dolzinaMm - n * slatWidth) / (n + 1)
   552→    return {
   553→      slatCount: n,
   554→      gap: Math.round(actualGap * 10) / 10,
   555→      compliant: actualGap <= 100,
   556→      postSpacing: dolzinaMm / Math.max(1, n > 5 ? Math.ceil(n / 5) : 2),
   557→    }
   558→  }

   589→    measurements.forEach((m) => {
   590→      if (m.segmentId) ids.add(m.segmentId)
   591→    })
   592→    return ids
   593→  }, [measurements])
   594→
   595→  const allSegments: Segment[] = useMemo(() => {
   596→    const known = new Map(segments.map((s) => [s.id, s]))
   597→    // dodaj segmente iz meritev, ki niso definirani
   598→    usedSegmentIds.forEach((id) => {
   599→      if (!known.has(id)) {
   600→        known.set(id, { id, name: id, type: 'ravni' })
   601→      }
   602→    })
   603→    return Array.from(known.values())
   604→  }, [segments, usedSegmentIds])
   605→
   606→  const segmentStats = useMemo(() => {
   607→    const stats = new Map<string, { totalLength: number; avgHeight: number; count: number }>()
   608→    allSegments.forEach((seg) => {
   609→      const segMeas = measurements.filter((m) => m.segmentId === seg.id)
   610→      const razdalje = segMeas.filter((m) => !m.tipMeritve || m.tipMeritve === 'RAZDALJA')
   611→      const visine = segMeas.filter((m) => m.tipMeritve === 'VISINA')
   612→      stats.set(seg.id, {
   613→        totalLength: razdalje.reduce((s, m) => s + m.dolzinaMm, 0),
   614→        avgHeight:
   615→          visine.length > 0
   616→            ? visine.reduce((s, m) => s + m.visinaMm, 0) / visine.length
   617→            : segMeas.length > 0
   618→              ? segMeas.reduce((s, m) => s + m.visinaMm, 0) / segMeas.length
   619→              : 0,
   620→        count: segMeas.length,
   621→      })
   622→    })
   623→    return stats
   624→  }, [allSegments, measurements])
   625→
   626→  // Grupiranje po datumu (obstoječa logika)
   627→  const groupedMeasurements = useMemo((): MeasurementGroup[] => {
   628→    const now = new Date()
   629→    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   630→    const yesterday = new Date(today.getTime() - 86400000)
   631→
   632→    const groups: MeasurementGroup[] = []
   633→    const grouped = new Map<string, Measurement[]>()
   634→
   635→    const sorted = [...measurements].sort(
   636→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   637→    )
   638→
   639→    for (const m of sorted) {
   640→      const mDate = new Date(m.createdAt)
   641→      const mDay = new Date(mDate.getFullYear(), mDate.getMonth(), mDate.getDate())
   642→
   643→      let label: string
   644→      if (mDay.getTime() === today.getTime()) {
   645→        label = 'Danes'
   646→      } else if (mDay.getTime() === yesterday.getTime()) {
   647→        label = 'Včeraj'
   648→      } else {
   649→        label = mDate.toLocaleDateString('sl-SI', { day: 'numeric', month: 'long' })
   650→      }
   651→
   652→      if (!grouped.has(label)) grouped.set(label, [])
   653→      grouped.get(label)!.push(m)
   654→    }
   655→
   656→    for (const [label, meas] of grouped) {
   657→      groups.push({ label, date: new Date(meas[0].createdAt), measurements: meas })
   658→    }
   659→
   660→    return groups
   661→  }, [measurements])

   947→  function renderRailingDiagram(dolzina: number, visina: number) {
